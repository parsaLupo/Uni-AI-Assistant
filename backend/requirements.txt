# Runtime dependencies for the University Smart Assistant API.
# Pinned deliberately: the stack is frozen for MVP by AGENTS.md section 1 and
# ADR-001; upgrades must be an explicit, documented change (see ADR-008).
#
# Versions verified against PyPI during Phase 00.

# LTS release (security support until April 2028) chosen over the newer 6.x
# major so the baseline does not absorb breaking changes mid-project.
Django==5.2.17
# 3.18.2 = security fix release (CVE-2026-114136/-114137/-114138, pip-audit
# Phase 15). 3.18.1 has no remaining advisories.
djangorestframework==3.18.2
django-cors-headers==4.9.0

# PostgreSQL driver (ADR-002). Binary wheel: no local compiler needed.
psycopg[binary]==3.3.6

# Production WSGI server (Phase 16). Gunicorn's gthread worker serves the
# streaming AI endpoints without blocking unrelated requests; configuration is
# in backend/docker/gunicorn.conf.py.
gunicorn==23.0.0

# OpenAPI generation — the committed openapi.yaml is the API contract
# (ADR-012, Phase 01). Installed dependency, documented per AGENTS rule 10.
drf-spectacular==0.30.0

# Authentication (ADR-014, Phase 03): JWT issue/verify for access tokens;
# rotation, reuse detection and cookies are project code in apps/accounts.
djangorestframework-simplejwt==5.5.1

# Task queue (ADR-011): Celery + Redis client for the document processing worker.
celery==5.6.3
redis==8.1.0
# Argon2id password hashing (docs/07 section 2, ADR-014 decision 6).
argon2-cffi==25.1.0

# PDF text extraction for document ingestion (Phase 08, RAG). PDFs are
# untrusted input, so this pin tracks the security fix stream: 6.21.4 clears
# the 2026 pypdf CVE list reported by pip-audit (Phase 15).
pypdf==6.21.4

# Spreadsheet reading/writing for the admin bulk-student import (Phase 16):
# openpyxl writes the official .xlsx template and reads registrar exports.
# Workbooks are opened read_only + data_only so formulas are never evaluated.
openpyxl==3.1.5
